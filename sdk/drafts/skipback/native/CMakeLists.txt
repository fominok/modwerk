cmake_minimum_required(VERSION 3.16)
project(skipback_native_probe CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OCTABAM_ROOT OR NOT EMU_BUILD)
  message(FATAL_ERROR "Set OCTABAM_ROOT to the pinned tooling checkout and EMU_BUILD to its built emulator")
endif()
add_executable(skipback_recorder_probe recorder_probe.cpp)
add_executable(skipback_core_probe core_probe.cpp)
foreach(TARGET skipback_recorder_probe skipback_core_probe)
target_include_directories(${TARGET} SYSTEM PRIVATE
  "${OCTABAM_ROOT}/tools/emu/ot_emu" "${OCTABAM_ROOT}/vendor")
target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Werror)
foreach(LIB libot_machine.a mc68k/lib68kEmu.a dsp56300/dsp56kEmu/libdsp56kEmu.a
    dsp56300/dsp56kBase/libdsp56kBase.a dsp56300/asmjit/libasmjit.a)
  if(NOT EXISTS "${EMU_BUILD}/${LIB}")
    message(FATAL_ERROR "Build the pinned ot_emu first; missing ${LIB}")
  endif()
  target_link_libraries(${TARGET} PRIVATE "${EMU_BUILD}/${LIB}")
endforeach()
target_link_libraries(${TARGET} PRIVATE pthread)
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/recorder_probe.cpp" PROBE_SHA)
target_compile_definitions(skipback_recorder_probe PRIVATE SB_PROBE_SHA="${PROBE_SHA}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core_probe.cpp" CORE_SHA)
target_compile_definitions(skipback_core_probe PRIVATE SB_PROBE_SHA="${CORE_SHA}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS recorder_probe.cpp core_probe.cpp)
